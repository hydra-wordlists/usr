sudo apt install -y mosquitto mosquitto-clients
sudo systemctl start/enable/status mosquitto

sudo iptables-save | sudo tee ~/iptables-backup.txt  //backup firewall rules to restore after configuration

t1:
mosquitto_sub -h localhost -p 1883 -t 'smartfactory/machine01/#' -v

t2:
mosquitto_pub -h localhost -p 1883 -t 'smartfactory/machine01/temperature' -m '24.00'

mosquitto_pub -h localhost -p 1883 -t 'smartfactory/machine01/temperature' -m '99.99'

mosquitto_pub -h localhost -p 1883 -t 'smartfactory/machine01/temperature' -m '24.00'


sudo iptables -I INPUT 1 -p tcp --dport 1883 -j DROP     //block all tcp traffic from unknown to port 1883
sudo iptables -I INPUT 1 -p tcp -s 10.0.2.0/24 --dport 1883 -j ACCEPT   //allow tcp traffic from this specific subnet    
sudo iptables-restore < ~/iptables-backup.txt
