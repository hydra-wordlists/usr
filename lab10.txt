#include <WiFi.h>
#include <PubSubClient.h>
#include "DHT.h"

#define DHTPIN 15
WiFiClient esp; PubSubClient mqtt(esp);
DHT dht(DHTPIN, DHT22);

void setup() {
  Serial.begin(115200); dht.begin();
  WiFi.begin("Wokwi-GUEST", "");
  while (WiFi.status() != WL_CONNECTED) delay(500);
  mqtt.setServer("test.mosquitto.org", 1883);
  while (!mqtt.connected()) { mqtt.connect("ESP32_Noel"); delay(1000); }
  mqtt.publish("unknown/status", "ONLINE");
}

void loop() {
  mqtt.loop();
  float t = dht.readTemperature(), h = dht.readHumidity();
  if (!isnan(t) && !isnan(h)) {
    char a[10], b[10];
    dtostrf(t, 1, 2, a); dtostrf(h, 1, 2, b);
    mqtt.publish("unknown/temperature", a);
    mqtt.publish("unknown/humidity", b);
    Serial.printf("Temp: %s C, Humidity: %s %%\n", a, b);
  }
  delay(3000);
}



big code


#include <WiFi.h>
#include <PubSubClient.h>
#include "DHT.h"

#define DHTPIN 15
#define DHTTYPE DHT22

const char* WIFI_SSID = "Wokwi-GUEST";
const char* MQTT_SERVER = "test.mosquitto.org";

const char* TEMP_TOPIC = "unknown/temperature";
const char* HUM_TOPIC = "unknown/humidity";
const char* STATUS_TOPIC = "unknown/status";

WiFiClient espClient;
PubSubClient mqtt(espClient);
DHT dht(DHTPIN, DHTTYPE);

unsigned long lastPublish = 0;

void setup() {
  Serial.begin(115200);
  dht.begin();

  WiFi.begin(WIFI_SSID, "");

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }

  Serial.println("WiFi connected!");

  mqtt.setServer(MQTT_SERVER, 1883);

  while (!mqtt.connected()) {
    if (mqtt.connect("unknown")) {
      Serial.println("MQTT connected!");
      mqtt.publish(STATUS_TOPIC, "ONLINE");
    } else {
      delay(2000);
    }
  }
}

void loop() {
  if (WiFi.status() != WL_CONNECTED) {
    WiFi.reconnect();
    delay(1000);
    return;
  }

  if (!mqtt.connected()) {
    if (!mqtt.connect("unknown")) {
      delay(2000);
      return;
    }
    mqtt.publish(STATUS_TOPIC, "ONLINE");
  }

  mqtt.loop();

  if (millis() - lastPublish >= 3000) {
    lastPublish = millis();

    float t = dht.readTemperature();
    float h = dht.readHumidity();

    if (isnan(t) || isnan(h)) {
      Serial.println("Sensor error!");
      return;
    }

    char temp[16], hum[16];
    snprintf(temp, sizeof(temp), "%.2f", t);
    snprintf(hum, sizeof(hum), "%.2f", h);

    mqtt.publish(TEMP_TOPIC, temp);
    mqtt.publish(HUM_TOPIC, hum);

    Serial.print("Temperature: ");
    Serial.print(temp);
    Serial.print(" C, Humidity: ");
    Serial.print(hum);
    Serial.println(" %");
  }
}
